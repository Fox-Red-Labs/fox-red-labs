cmake_minimum_required(VERSION 3.20.0)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(core1_rpmsg)
target_sources(app PRIVATE src/main.c)
target_sources_ifdef(CONFIG_FRL_RING_CONSOLE app PRIVATE src/ring_console.c)
# shared layout header; the build host's copy of this repo's am62l-evm/rpmsg folder (RPMSG_DIR, default ~/am62l-zephyr-setup/rpmsg)
if(DEFINED ENV{RPMSG_DIR})
  target_include_directories(app PRIVATE $ENV{RPMSG_DIR})
else()
  target_include_directories(app PRIVATE $ENV{HOME}/am62l-zephyr-setup/rpmsg)
endif()
